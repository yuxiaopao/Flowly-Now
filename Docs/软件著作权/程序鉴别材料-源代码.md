"""
NL to Workflow service for generating workflows from natural language.
"""

import json
import re
import logging
import time
import uuid
from typing import Dict, Any, Optional, List, Generator, Tuple
from django.conf import settings

from gateway.providers.registry import get_registry, register_default_providers, ProviderRegistry
from .constants import (
    DEFAULT_TEMPERATURE,
    DEFAULT_MAX_TOKENS,
    FALLBACK_MODEL,
    ERROR_MESSAGES,
    VALIDATION_MESSAGES,
    MODIFICATION_KEYWORDS,
    SUGGESTION_EXTRACTION_PATTERNS,
    SUPPORTED_NODE_TYPES,
    SUGGESTION_TEMPLATES,
    MAX_SUGGESTIONS,
)
from .output_parser import WorkflowOutputParser
from .prompts import (
    WorkflowType,
    get_workflow_generation_prompt,
    format_workflow_prompt,
    get_modification_prompt,
)

logger = logging.getLogger(__name__)

# Module-level constant for the workflow generation prompt template
FALLBACK_WORKFLOW_TEMPLATE = """
## 工作流结构要求

### 必须包含的节点
1. **start**: 开始节点 - 每个工作流只能有一个开始节点
2. **end**: 结束节点 - 标记工作流结束

### 可选节点类型（根据需求选择）
{node_types}

### 输出格式（必须是有效JSON）
请严格按照以下JSON格式输出：

{
    "name": "工作流名称",
    "description": "工作流功能描述",
    "nodes": [
        {
            "id": "node_1",
            "type": "节点类型",
            "name": "节点名称",
            "position": {"x": 数字, "y": 数字},
            "config": {}
        }
    ],
    "edges": [
        {
            "id": "edge_1",
            "source": "源节点ID",
            "target": "目标节点ID",
            "label": "连线标签"
        }
    ]
}
"""

class NLToWorkflowService:
    """
    Service for converting natural language descriptions to workflows.
    
    Features:
    - Intent analysis and classification
    - Workflow structure generation
    - Multi-round conversation support
    - Template recommendation
    - Streaming output support
    """

    def __init__(self, model: Optional[str] = None, temperature: float = DEFAULT_TEMPERATURE):
        """Initialize the NL to Workflow service."""
        self.model = model or FALLBACK_MODEL
        self.temperature = temperature
        self.max_tokens = DEFAULT_MAX_TOKENS
        self.parser = WorkflowOutputParser()
        self._registry = None

    @property
    def registry(self):
        """Lazy initialization of provider registry."""
        if self._registry is None:
            try:
                register_default_providers()
                self._registry = get_registry()
            except Exception as e:
                logger.warning(f"Failed to initialize registry: {e}")
                self._registry = None
        return self._registry

    def analyze_intent(self, description: str) -> Dict[str, Any]:
        """
        Analyze user intent from description.
        
        Args:
            description: User's natural language description
            
        Returns:
            Dict containing intent analysis results
        """
        intent_prompt = f"""分析以下工作流需求，确定其类型和关键特征：

需求：{description}

请返回JSON格式的分析结果：
{{
    "workflow_type": "类型（qa_bot/data_processing/translation/moderation/article_generation/customer_service/generic）",
    "required_nodes": ["需要的节点类型列表"],
    "complexity": "简单/中等/复杂",
    "key_features": ["关键特性列表"],
    "suggested_tools": ["建议使用的工具列表"]
}}
"""
        try:
            response = self._call_llm(intent_prompt)
            return json.loads(response)
        except Exception as e:
            logger.error(f"Intent analysis failed: {e}")
            return {
                "workflow_type": "generic",
                "required_nodes": ["llm"],
                "complexity": "中等",
                "key_features": [],
                "suggested_tools": []
            }

    def generate_workflow(
        self,
        description: str,
        intent: Optional[Dict[str, Any]] = None,
        context: Optional[List[Dict]] = None
    ) -> Dict[str, Any]:
        """
        Generate workflow from natural language description.
        
        Args:
            description: User's natural language description
            intent: Pre-analyzed intent (optional)
            context: Conversation history (optional)
            
        Returns:
            Generated workflow structure
        """
        # Analyze intent if not provided
        if intent is None:
            intent = self.analyze_intent(description)
        
        # Get workflow type
        workflow_type = self._get_workflow_type(intent)
        
        # Build prompt
        prompt = self._build_generation_prompt(
            description,
            workflow_type,
            context
        )
        
        # Call LLM
        try:
            response = self._call_llm(prompt)
            workflow = self.parser.parse(response)
            return workflow
        except Exception as e:
            logger.error(f"Workflow generation failed: {e}")
            return self._generate_fallback_workflow(description, intent)

    def _call_llm(self, prompt: str) -> str:
        """Call LLM with prompt."""
        provider = None
        if self.registry:
            provider = self.registry.get('openai')
        
        if provider:
            response = provider.chat(
                messages=[{"role": "user", "content": prompt}],
                model=self.model,
                temperature=self.temperature,
                max_tokens=self.max_tokens
            )
            return response.content
        else:
            raise RuntimeError("No LLM provider available")

    def _get_workflow_type(self, intent: Dict[str, Any]) -> WorkflowType:
        """Get workflow type from intent analysis."""
        type_mapping = {
            'qa_bot': WorkflowType.QA_BOT,
            'data_processing': WorkflowType.DATA_PROCESSING,
            'translation': WorkflowType.TRANSLATION,
            'moderation': WorkflowType.MODERATION,
            'article_generation': WorkflowType.ARTICLE_GENERATION,
            'customer_service': WorkflowType.CUSTOMER_SERVICE,
        }
        type_str = intent.get('workflow_type', 'generic')
        return type_mapping.get(type_str, WorkflowType.GENERIC)

    def _build_generation_prompt(
        self,
        description: str,
        workflow_type: WorkflowType,
        context: Optional[List[Dict]] = None
    ) -> str:
        """Build prompt for workflow generation."""
        prompt_template = get_workflow_generation_prompt(workflow_type)
        
        # Format prompt with description
        prompt = prompt_template.format(description=description)
        
        # Add conversation context if available
        if context:
            context_str = "\n".join([
                f"{msg.get('role', 'user')}: {msg.get('content', '')}"
                for msg in context[-5:]
            ])
            prompt = f"对话上下文：\n{context_str}\n\n当前需求：{description}"
        
        return prompt

    def _generate_fallback_workflow(
        self,
        description: str,
        intent: Dict[str, Any]
    ) -> Dict[str, Any]:
        """Generate a simple fallback workflow when LLM fails."""
        return {
            "name": f"工作流_{int(time.time())}",
            "description": description,
            "nodes": [
                {
                    "id": "node_1",
                    "type": "start",
                    "name": "开始",
                    "position": {"x": 100, "y": 100},
                    "config": {}
                },
                {
                    "id": "node_2",
                    "type": "input",
                    "name": "输入",
                    "position": {"x": 280, "y": 100},
                    "config": {"input_name": "user_input"}
                },
                {
                    "id": "node_3",
                    "type": "llm",
                    "name": "AI处理",
                    "position": {"x": 460, "y": 100},
                    "config": {
                        "model": self.model,

                    }
                },

                    "id": "node_4",
                    "type": "end",
                    "name": "结束",

                {"id": "edge_2", "source": "node_2", "target": "node_3", "label": ""},
                {"id": "edge_3", "source": "node_3", "target": "node_4", "label": ""}
            ]
        }

    def validate_workflow(self, workflow: Dict[str, Any]) -> Tuple[bool, List[str]]:
        """
        Validate generated workflow.
        
        Args:
            workflow: Workflow structure to validate
            
        Returns:
            Tuple of (is_valid, error_messages)
        """
        errors = []
        
        # Check required fields
        if 'name' not in workflow:
            errors.append(VALIDATION_MESSAGES['missing_name'])
        if 'nodes' not in workflow:
            errors.append(VALIDATION_MESSAGES['missing_nodes'])
        if 'edges' not in workflow:
            errors.append(VALIDATION_MESSAGES['missing_edges'])
        
        # Check nodes
        if 'nodes' in workflow:
            node_ids = set()
            has_start = False
            has_end = False
            
            for node in workflow['nodes']:
                # Check node ID
                if 'id' not in node:
                    errors.append(VALIDATION_MESSAGES['missing_node_id'])
                else:
                    if node['id'] in node_ids:
                        errors.append(VALIDATION_MESSAGES['duplicate_node_id'].format(node['id']))
                    node_ids.add(node['id'])
                
                # Check node type
                if 'type' not in node:
                    errors.append(VALIDATION_MESSAGES['missing_node_type'])
                else:
                    if node['type'] == 'start':
                        has_start = True
                    if node['type'] == 'end':
                        has_end = True
            
            # Validate start and end nodes
            if not has_start:
                errors.append(VALIDATION_MESSAGES['missing_start_node'])
            if not has_end:
                errors.append(VALIDATION_MESSAGES['missing_end_node'])
        
        # Check edges
        if 'edges' in workflow and 'nodes' in workflow:
            node_ids = {n['id'] for n in workflow['nodes'] if 'id' in n}
            
            for edge in workflow['edges']:
                if 'source' not in edge:
                    errors.append(VALIDATION_MESSAGES['missing_edge_source'])
                elif edge['source'] not in node_ids:
                    errors.append(VALIDATION_MESSAGES['invalid_edge_source'].format(edge['source']))
                
                if 'target' not in edge:
                    errors.append(VALIDATION_MESSAGES['missing_edge_target'])
                elif edge['target'] not in node_ids:
                    errors.append(VALIDATION_MESSAGES['invalid_edge_target'].format(edge['target']))
        
        return len(errors) == 0, errors

    def get_suggestions(self, workflow: Dict[str, Any]) -> List[str]:
        """
        Get modification suggestions for a workflow.
        
        Args:
            workflow: Current workflow structure
            
        Returns:
            List of suggested improvements
        """
        suggestions = []
        
        # Check complexity
        node_count = len(workflow.get('nodes', []))
        if node_count < 3:
            suggestions.append(SUGGESTION_TEMPLATES['add_nodes'])
        
        # Check for error handling
        has_llm_nodes = any(
            n.get('type') == 'llm' for n in workflow.get('nodes', [])
        )
        has_condition = any(
            n.get('type') == 'condition' for n in workflow.get('nodes', [])
        )
        
        if has_llm_nodes and not has_condition:
            suggestions.append(SUGGESTION_TEMPLATES['add_condition'])
        
        # Check for input validation
        has_input = any(
            n.get('type') == 'input' for n in workflow.get('nodes', [])
        )
        if not has_input:
            suggestions.append(SUGGESTION_TEMPLATES['add_input'])
        
        # Limit suggestions
        return suggestions[:MAX_SUGGESTIONS]


class WorkflowOutputParser:
    """
    Parser for LLM-generated workflow outputs.
    
    Features:
    - JSON extraction
    - Schema validation
    - Error recovery
    """

    def __init__(self):
        """Initialize the parser."""
        self.validation_rules = self._build_validation_rules()

    def parse(self, raw_output: str) -> Dict[str, Any]:
        """
        Parse raw LLM output into workflow structure.
        
        Args:
            raw_output: Raw text output from LLM
            
        Returns:
            Parsed workflow structure
        """
        # Try to extract JSON
        json_str = self._extract_json(raw_output)
        
        if json_str:
            try:
                workflow = json.loads(json_str)
                return self._normalize_workflow(workflow)
            except json.JSONDecodeError as e:
                logger.warning(f"JSON parse failed: {e}")
        
        # Try to fix common errors
        fixed_output = self._fix_common_errors(raw_output)
        json_str = self._extract_json(fixed_output)
        
        if json_str:
            try:
                workflow = json.loads(json_str)
                return self._normalize_workflow(workflow)
            except json.JSONDecodeError:
                pass
        
        # Return empty workflow
        return self._get_empty_workflow()

    def _extract_json(self, text: str) -> Optional[str]:
        """Extract JSON from text."""
        # Try markdown code blocks
        match = re.search(r'```(?:json)?\s*([\s\S]*?)\s*```', text)
        if match:
            return match.group(1)
        
        # Try bare JSON
        match = re.search(r'\{[\s\S]*\}', text)
        if match:
            return match.group(0)
        
        return None

    def _fix_common_errors(self, text: str) -> str:
        """Fix common LLM output errors."""
        # Remove trailing commas
        text = re.sub(r',\s*\}', '}', text)
        text = re.sub(r',\s*\]', ']', text)
        
        # Fix single quotes to double quotes
        text = re.sub(r"'([^']*)':", r'"\1":', text)
        
        return text

    def _normalize_workflow(self, workflow: Dict[str, Any]) -> Dict[str, Any]:
        """Normalize workflow structure."""
        normalized = {
            'name': workflow.get('name', '未命名工作流'),
            'description': workflow.get('description', ''),
            'nodes': [],
            'edges': []
        }
        
        # Normalize nodes
        for node in workflow.get('nodes', []):
            normalized_node = {
                'id': node.get('id', f"node_{uuid.uuid4().hex[:8]}"),
                'type': node.get('type', 'llm'),
                'name': node.get('name', '节点'),
                'position': node.get('position', {'x': 0, 'y': 0}),
                'config': node.get('config', {})
            }
            normalized['nodes'].append(normalized_node)
        
        # Normalize edges
        for edge in workflow.get('edges', []):
            normalized_edge = {
                'id': edge.get('id', f"edge_{uuid.uuid4().hex[:8]}"),
                'source': edge.get('source', ''),
                'target': edge.get('target', ''),
                'label': edge.get('label', '')
            }
            normalized['edges'].append(normalized_edge)
        
        return normalized

    def _get_empty_workflow(self) -> Dict[str, Any]:
        """Get empty workflow structure."""
        return {
            'name': '空工作流',
            'description': '',
            'nodes': [
                {
                    'id': 'node_1',
                    'type': 'start',
                    'name': '开始',
                    'position': {'x': 100, 'y': 100},
                    'config': {}
                },
                {
                    'id': 'node_2',
                    'type': 'end',
                    'name': '结束',
                    'position': {'x': 300, 'y': 100},
                    'config': {}
                }
            ],
            'edges': []
        }

    def _build_validation_rules(self) -> Dict[str, Any]:
        """Build validation rules for workflow."""
        return {
            'required_fields': ['name', 'nodes', 'edges'],
            'node_types': list(SUPPORTED_NODE_TYPES.keys()),
            'min_nodes': 2,
            'max_nodes': 1000
        }

/**
 * NL Workflow Store - Natural Language to Workflow
 *
 * Manages NL-to-workflow conversation state.
 * Used by components that need centralized state management.

import { defineStore } from 'pinia'
import { nlWorkflowApi } from '@/api/nl-workflow'

export const useNLWorkflowStore = defineStore('nlWorkflow', {
  state: () => ({
    // Current conversation
    messages: [],

    // Generation status
    isGenerating: false,
    generationError: null,
    streamingProgress: 0,
    streamingStage: '',
    isStreaming: false,

    // Generated workflow
    generatedWorkflow: null,

    // Intent analysis
    lastIntent: null,

    // History
    history: [],
    historyLoading: false,

    // Available models
    models: [],
    selectedModel: null,
    modelsLoading: false,

    // Current session
    currentSessionId: null,

    // Generation settings
    settings: {
      includeDescription: true,
      includeValidation: true,
      temperature: 0.7,
      maxTokens: 2048
    }
  }),

  getters: {
    // Get conversation history for API
    conversationHistory: (state) => {
      return state.messages.map(msg => ({
        role: msg.role,
        content: msg.content
      }))
    },

    // Get last user message
    lastUserMessage: (state) => {
      const userMessages = state.messages.filter(m => m.role === 'user')
      return userMessages[userMessages.length - 1] || null
    },

    // Check if can continue
    canContinue: (state) => {
      return state.generatedWorkflow !== null && !state.isGenerating
    },

    // Get message count
    messageCount: (state) => state.messages.length,

    // Check if conversation is empty
    isEmpty: (state) => state.messages.length === 0,

    // Has active session
    hasSession: (state) => state.currentSessionId !== null
  },

  actions: {
    /**
     * Create a new session
     */
    async createSession(params = {}) {
      try {
        const response = await nlWorkflowApi.createSession({
          name: params.name,
          description: params.description,
          model: this.selectedModel
        })
        this.currentSessionId = response.id
        this.messages = []
        this.generatedWorkflow = null
        return response
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Load session history
     */
    async loadSession(sessionId) {
      try {
        const response = await nlWorkflowApi.getSession(sessionId)
        this.currentSessionId = sessionId
        this.messages = response.messages || []
        this.generatedWorkflow = response.generated_workflow
        this.selectedModel = response.model
        return response
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Send message and generate response
     */
    async sendMessage(content) {
      if (!content.trim()) return

      // Add user message
      this.messages.push({
        role: 'user',
        content,
        timestamp: Date.now()
      })

      this.isGenerating = true
      this.generationError = null

      try {
        const response = await nlWorkflowApi.sendMessage(
          this.currentSessionId,
          { message: content }
        )

        // Add assistant response
        if (response.message) {
          this.messages.push({
            role: 'assistant',
            content: response.message.content,
            timestamp: Date.now()
          })
        }

        // Update generated workflow
        if (response.generated_workflow) {
          this.generatedWorkflow = response.generated_workflow
        }

        return response
      } catch (error) {
        this.generationError = error.message
        throw error
      } finally {
        this.isGenerating = false
      }
    },

    /**
     * Stream message response
     */
    async streamMessage(content, onChunk) {
      if (!content.trim()) return

      // Add user message
      this.messages.push({
        role: 'user',
        content,
        timestamp: Date.now()
      })

      this.isGenerating = true
      this.isStreaming = true
      this.generationError = null

      try {
        let fullContent = ''

        await nlWorkflowApi.streamMessage(
          this.currentSessionId,
          { message: content },
          (chunk) => {
            fullContent += chunk
            this.streamingProgress = Math.min(
              (fullContent.length / 1000) * 100,
              95
            )
            this.streamingStage = '生成中...'

            if (onChunk) {
              onChunk(chunk, fullContent)
            }
          }
        )

        // Add complete response
        this.messages.push({
          role: 'assistant',
          content: fullContent,
          timestamp: Date.now()
        })

        this.streamingProgress = 100
        this.streamingStage = '完成'

        return fullContent
      } catch (error) {
        this.generationError = error.message
        throw error
      } finally {
        this.isGenerating = false
        this.isStreaming = false
        this.streamingProgress = 0
        this.streamingStage = ''
      }
    },

    /**
     * Generate workflow directly from description
     */
    async generateWorkflow(description) {
      this.isGenerating = true
      this.generationError = null

      try {
        const response = await nlWorkflowApi.generateWorkflow({
          description,
          model: this.selectedModel,
          temperature: this.settings.temperature,
          max_tokens: this.settings.maxTokens
        })

        this.generatedWorkflow = response.workflow
        this.lastIntent = response.intent

        return response.workflow
      } catch (error) {
        this.generationError = error.message
        throw error
      } finally {
        this.isGenerating = false
      }
    },

    /**
     * Preview workflow without saving
     */
    async previewWorkflow(description) {
      try {
        return await nlWorkflowApi.previewWorkflow({ description })
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Confirm and save generated workflow
     */
    async confirmWorkflow() {
      if (!this.generatedWorkflow) {
        throw new Error('No workflow to confirm')
      }

      try {
        const response = await nlWorkflowApi.confirmSession(this.currentSessionId)
        return response
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Analyze intent
     */
    async analyzeIntent(description) {
      try {
        const response = await nlWorkflowApi.analyzeIntent({ description })
        this.lastIntent = response
        return response
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Get suggestions for current workflow
     */
    async getSuggestions() {
      if (!this.generatedWorkflow) {
        return []
      }

      try {
        return await nlWorkflowApi.getSuggestions({
          workflow: this.generatedWorkflow
        })
      } catch (error) {
        console.error('Failed to get suggestions:', error)
        return []
      }
    },

    /**
     * Load available models
     */
    async loadModels() {
      this.modelsLoading = true
      try {
        const response = await nlWorkflowApi.getModels()
        this.models = response.models || []
        if (this.models.length > 0 && !this.selectedModel) {
          this.selectedModel = this.models[0].id
        }
        return this.models
      } catch (error) {
        console.error('Failed to load models:', error)
        return []
      } finally {
        this.modelsLoading = false
      }
    },

    /**
     * Load session history
     */
    async loadHistory(params = {}) {
      this.historyLoading = true
      try {
        const response = await nlWorkflowApi.getSessions(params)
        this.history = response.results || []
        return this.history
      } catch (error) {
        console.error('Failed to load history:', error)
        return []
      } finally {
        this.historyLoading = false
      }
    },

    /**
     * Delete session
     */
    async deleteSession(sessionId) {
      try {
        await nlWorkflowApi.deleteSession(sessionId)
        this.history = this.history.filter(s => s.id !== sessionId)
        if (this.currentSessionId === sessionId) {
          this.reset()
        }
      } catch (error) {
        this.generationError = error.message
        throw error
      }
    },

    /**
     * Reset store state
     */
    reset() {
      this.messages = []
      this.generatedWorkflow = null
      this.lastIntent = null
      this.currentSessionId = null
      this.isGenerating = false
      this.generationError = null
      this.streamingProgress = 0
      this.streamingStage = ''
      this.isStreaming = false
    },

    /**
     * Update settings
     */
    updateSettings(newSettings) {
      this.settings = { ...this.settings, ...newSettings }
    }
  }
})
